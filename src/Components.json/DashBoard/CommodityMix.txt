import React, { useEffect, useState } from "react";
import { Bar, BarChart, Cell, ResponsiveContainer, Tooltip, XAxis, YAxis } from "recharts";
import ChartCard from "@/components/dashboard/ChartCard";
import { commodityVolumeSeries, fetchAll } from "@/lib/analytics";
import { fmtNum } from "@/lib/format";

const COLORS = ["#14503C", "#1F6B51", "#4B8B5A", "#7FA88A", "#C79B4B", "#D9BE86"];

export default function CommodityMix() {
  const [series, setSeries] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchAll("CommodityBatch").then((batches) => {
      setSeries(commodityVolumeSeries(batches).map((item) => ({ ...item, name: item.name })));
      setLoading(false);
    });
  }, []);

  return (
    <ChartCard title="Commodity distribution" subtitle="Recorded volume by commodity (kg)">
      {loading ? (
        <div className="h-[252px] animate-pulse rounded-lg bg-muted/60" />
      ) : series.length === 0 ? (
        <div className="flex h-[252px] items-center justify-center text-[12.5px] text-muted-foreground">
          No batches recorded yet.
        </div>
      ) : (
        <ResponsiveContainer width="100%" height={252}>
          <BarChart data={series} layout="vertical" margin={{ top: 0, right: 12, left: 4, bottom: 0 }}>
            <XAxis type="number" hide />
            <YAxis
              type="category"
              dataKey="name"
              width={72}
              tick={{ fontSize: 10.5, fill: "#3C4A44" }}
              axisLine={false}
              tickLine={false}
            />
            <Tooltip
              cursor={{ fill: "#F8F4EC" }}
              contentStyle={{ borderRadius: 10, border: "1px solid #E7E1D5", fontSize: 11.5 }}
              formatter={(value) => [fmtNum(value), "Volume (kg)"]}
            />
            <Bar dataKey="volume" radius={[0, 4, 4, 0]} barSize={16}>
              {series.map((entry, index) => (
                <Cell key={entry.name} fill={COLORS[index % COLORS.length]} />
              ))}
            </Bar>
          </BarChart>
        </ResponsiveContainer>
      )}
    </ChartCard>
  );
}
