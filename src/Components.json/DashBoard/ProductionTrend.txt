import React, { useEffect, useState } from "react";
import { Area, AreaChart, CartesianGrid, ResponsiveContainer, Tooltip, XAxis, YAxis } from "recharts";
import ChartCard from "@/components/dashboard/ChartCard";
import { fetchAll, monthlyVolumeSeries } from "@/lib/analytics";
import { fmtNum } from "@/lib/format";

export default function ProductionTrend() {
  const [series, setSeries] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchAll("CommodityBatch").then((batches) => {
      setSeries(monthlyVolumeSeries(batches));
      setLoading(false);
    });
  }, []);

  return (
    <ChartCard
      title="Agricultural production trend"
      subtitle="Recorded batch volume by harvest month (kg)"
      className="lg:col-span-2"
    >
      {loading ? (
        <div className="h-[252px] animate-pulse rounded-lg bg-muted/60" />
      ) : series.length === 0 ? (
        <div className="flex h-[252px] items-center justify-center text-[12.5px] text-muted-foreground">
          No batches recorded yet.
        </div>
      ) : (
        <ResponsiveContainer width="100%" height={252}>
          <AreaChart data={series} margin={{ top: 4, right: 8, left: -16, bottom: 0 }}>
            <defs>
              <linearGradient id="volumeFill" x1="0" y1="0" x2="0" y2="1">
                <stop offset="0%" stopColor="#14503C" stopOpacity={0.22} />
                <stop offset="100%" stopColor="#14503C" stopOpacity={0.02} />
              </linearGradient>
            </defs>
            <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#E7E1D5" />
            <XAxis
              dataKey="label"
              tick={{ fontSize: 10.5, fill: "#6B7C74" }}
              axisLine={false}
              tickLine={false}
            />
            <YAxis
              tick={{ fontSize: 10.5, fill: "#6B7C74" }}
              axisLine={false}
              tickLine={false}
              tickFormatter={(value) => (value >= 1000 ? `${value / 1000}k` : value)}
            />
            <Tooltip
              contentStyle={{
                borderRadius: 10,
                border: "1px solid #E7E1D5",
                fontSize: 11.5,
                boxShadow: "0 12px 30px -18px rgba(12,53,39,0.4)",
              }}
              formatter={(value, name) => [fmtNum(value), name === "volume" ? "Volume (kg)" : name]}
            />
            <Area
              type="monotone"
              dataKey="volume"
              stroke="#14503C"
              strokeWidth={2}
              fill="url(#volumeFill)"
              dot={{ r: 2.5, fill: "#14503C", strokeWidth: 0 }}
              activeDot={{ r: 4, fill: "#C79B4B", strokeWidth: 0 }}
            />
          </AreaChart>
        </ResponsiveContainer>
      )}
    </ChartCard>
  );
}
