import React, { useEffect, useState } from "react";
import { Link } from "react-router-dom";
import ChartCard from "@/components/dashboard/ChartCard";
import { fetchAll } from "@/lib/analytics";
import { fmtDate } from "@/lib/format";

const SEVERITY = {
  critical: "border-[#B4553A]/30 bg-[#F8E8E2] text-[#97402A]",
  high: "border-[#C98A5A]/30 bg-[#FAEEE2] text-[#9A5B26]",
  moderate: "border-gold/30 bg-gold-light text-gold-dark",
  low: "border-border bg-muted/70 text-muted-foreground",
};

export default function RiskPanel() {
  const [alerts, setAlerts] = useState(null);

  useEffect(() => {
    fetchAll("RiskAlert", { status: { $in: ["open", "monitoring"] } }, 20).then((records) => {
      setAlerts(records.slice(0, 4));
    });
  }, []);

  return (
    <ChartCard
      title="Emerging risk alerts"
      subtitle="Open and monitored signals requiring verification"
      action={
        <Link to="/alerts" className="text-[11px] font-semibold text-forest underline-offset-4 hover:underline">
          All alerts
        </Link>
      }
    >
      {!alerts ? (
        <div className="h-[200px] animate-pulse rounded-lg bg-muted/60" />
      ) : alerts.length === 0 ? (
        <div className="flex h-[160px] items-center justify-center text-[12.5px] text-muted-foreground">
          No active alerts.
        </div>
      ) : (
        <ul className="space-y-2.5">
          {alerts.map((alert) => (
            <li key={alert.id} className="rounded-lg border border-border/70 p-3">
              <div className="flex items-center justify-between gap-2">
                <span
                  className={`rounded-full border px-2 py-0.5 text-[9px] font-semibold uppercase tracking-[0.1em] ${
                    SEVERITY[alert.severity] || SEVERITY.low
                  }`}
                >
                  {alert.severity}
                </span>
                <span className="text-[10px] text-muted-foreground">{fmtDate(alert.detected_date)}</span>
              </div>
              <p className="mt-2 text-[12.5px] font-medium leading-snug text-forest-dark">{alert.title}</p>
              <p className="mt-1 text-[11px] text-muted-foreground">
                {[alert.state, `${alert.confidence || "medium"} confidence`].filter(Boolean).join(" · ")}
              </p>
            </li>
          ))}
        </ul>
      )}
    </ChartCard>
  );
}
