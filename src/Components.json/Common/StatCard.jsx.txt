import React from "react";
import { cn } from "@/lib/utils";
import { fmtNum } from "@/lib/format";

export default function StatCard({ label, value, sub, loading, tone = "forest" }) {
  const tones = {
    forest: "text-forest-dark",
    moss: "text-moss",
    gold: "text-gold-dark",
    clay: "text-[#A6552F]",
  };
  return (
    <div className="surface group relative overflow-hidden p-5 transition-colors duration-300 hover:border-forest/25">
      <span className="absolute left-0 top-0 h-full w-[3px] bg-gradient-to-b from-forest/70 to-moss/40 opacity-0 transition-opacity duration-300 group-hover:opacity-100" />
      <p className="label-caps">{label}</p>
      <p className={cn("tabular mt-3 text-[30px] font-semibold leading-none tracking-tight", tones[tone])}>
        {loading ? <span className="inline-block h-7 w-16 animate-pulse rounded bg-muted" /> : fmtNum(value)}
      </p>
      {sub && <p className="mt-2 text-xs leading-relaxed text-muted-foreground">{sub}</p>}
    </div>
  );
}
