import React, { useEffect, useState } from "react";
import { base44 } from "@/api/base44Client";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
import { COMMODITIES, NIGERIAN_STATES } from "@/lib/constants";

const EMPTY = {
  name: "",
  record_type: "cooperative",
  state: "",
  lga: "",
  community: "",
  member_count: "",
  production_capacity_t: "",
  primary_commodities: [],
  coordinator_name: "",
  contact_phone: "",
  aggregation_point: "",
  status: "active",
  latitude: "",
  longitude: "",
  notes: "",
};

export default function CooperativeForm({ open, onOpenChange, cooperative, onSaved }) {
  const [form, setForm] = useState(EMPTY);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    if (!open) return;
    setForm(cooperative ? { ...EMPTY, ...cooperative } : EMPTY);
    setError("");
  }, [open, cooperative]);

  const set = (field) => (event) => setForm({ ...form, [field]: event.target.value });
  const numeric = (field) => (event) =>
    setForm({ ...form, [field]: event.target.value === "" ? "" : Number(event.target.value) });

  const toggleCommodity = (item) =>
    setForm({
      ...form,
      primary_commodities: form.primary_commodities?.includes(item)
        ? form.primary_commodities.filter((value) => value !== item)
        : [...(form.primary_commodities || []), item],
    });

  const submit = async (event) => {
    event.preventDefault();
    setSaving(true);
    setError("");
    try {
      const payload = {
        ...form,
        member_count: form.member_count === "" ? null : Number(form.member_count),
        production_capacity_t: form.production_capacity_t === "" ? null : Number(form.production_capacity_t),
        latitude: form.latitude === "" ? null : Number(form.latitude),
        longitude: form.longitude === "" ? null : Number(form.longitude),
      };
      delete payload.id;
      delete payload.created_date;
      delete payload.updated_date;
      delete payload.created_by_id;
      if (cooperative) await base44.entities.Cooperative.update(cooperative.id, payload);
      else await base44.entities.Cooperative.create({ ...payload, demo_record: false });
      onSaved();
      onOpenChange(false);
    } catch (err) {
      setError("The record could not be saved. Please check the required fields.");
    } finally {
      setSaving(false);
    }
  };

  const remove = async () => {
    setSaving(true);
    await base44.entities.Cooperative.delete(cooperative.id);
    setSaving(false);
    onSaved();
    onOpenChange(false);
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-h-[92vh] max-w-2xl overflow-y-auto">
        <DialogHeader>
          <DialogTitle className="font-display text-2xl text-forest-dark">
            {cooperative ? "Update group record" : "Register cooperative or cluster"}
          </DialogTitle>
          <DialogDescription className="text-[12px]">
            Cooperatives group individual farmers; clusters describe a shared production geography.
          </DialogDescription>
        </DialogHeader>

        <form onSubmit={submit} className="mt-4 space-y-4">
          <div>
            <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
              Record type
            </Label>
            <div className="mt-2 flex gap-2">
              {[
                { value: "cooperative", label: "Cooperative" },
                { value: "cluster", label: "Agricultural cluster" },
              ].map((option) => (
                <button
                  key={option.value}
                  type="button"
                  onClick={() => setForm({ ...form, record_type: option.value })}
                  className={`rounded-lg border px-3.5 py-2 text-[12px] font-medium transition-colors ${
                    form.record_type === option.value
                      ? "border-forest bg-forest-tint text-forest-dark"
                      : "border-border bg-white text-foreground/70"
                  }`}
                >
                  {option.label}
                </button>
              ))}
            </div>
          </div>

          <div className="grid gap-3.5 sm:grid-cols-2">
            <div className="sm:col-span-2">
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Name
              </Label>
              <Input required className="mt-1.5" value={form.name} onChange={set("name")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                State
              </Label>
              <select required value={form.state} onChange={set("state")} className="select-input mt-1.5">
                <option value="">Select state</option>
                {NIGERIAN_STATES.map((state) => (
                  <option key={state}>{state}</option>
                ))}
              </select>
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                LGA
              </Label>
              <Input className="mt-1.5" value={form.lga} onChange={set("lga")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Community
              </Label>
              <Input className="mt-1.5" value={form.community} onChange={set("community")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Aggregation point
              </Label>
              <Input className="mt-1.5" value={form.aggregation_point} onChange={set("aggregation_point")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Members
              </Label>
              <Input type="number" className="mt-1.5" value={form.member_count} onChange={numeric("member_count")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Production capacity (t)
              </Label>
              <Input
                type="number"
                className="mt-1.5"
                value={form.production_capacity_t}
                onChange={numeric("production_capacity_t")}
              />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Coordinator
              </Label>
              <Input className="mt-1.5" value={form.coordinator_name} onChange={set("coordinator_name")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Contact phone
              </Label>
              <Input className="mt-1.5" value={form.contact_phone} onChange={set("contact_phone")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Status
              </Label>
              <select value={form.status} onChange={set("status")} className="select-input mt-1.5">
                <option value="active">Active</option>
                <option value="inactive">Inactive</option>
              </select>
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Latitude
              </Label>
              <Input type="number" step="0.0001" className="mt-1.5" value={form.latitude} onChange={numeric("latitude")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Longitude
              </Label>
              <Input type="number" step="0.0001" className="mt-1.5" value={form.longitude} onChange={numeric("longitude")} />
            </div>
          </div>

          <div>
            <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
              Primary commodities
            </Label>
            <div className="mt-2 flex flex-wrap gap-1.5">
              {COMMODITIES.map((item) => {
                const active = form.primary_commodities?.includes(item);
                return (
                  <button
                    key={item}
                    type="button"
                    onClick={() => toggleCommodity(item)}
                    className={`rounded-full border px-3 py-1 text-[11.5px] transition-colors ${
                      active
                        ? "border-forest bg-forest text-primary-foreground"
                        : "border-border bg-white text-foreground/70 hover:border-forest/40"
                    }`}
                  >
                    {item}
                  </button>
                );
              })}
            </div>
          </div>

          <div>
            <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
              Notes
            </Label>
            <Textarea rows={2} className="mt-1.5 resize-none" value={form.notes} onChange={set("notes")} />
          </div>

          {error && <p className="text-[12px] text-destructive">{error}</p>}

          <div className="flex items-center justify-between gap-2 pt-1">
            {cooperative ? (
              <Button type="button" variant="ghost" className="text-destructive" onClick={remove} disabled={saving}>
                Delete record
              </Button>
            ) : (
              <span />
            )}
            <div className="flex gap-2">
              <Button type="button" variant="ghost" onClick={() => onOpenChange(false)}>
                Cancel
              </Button>
              <Button type="submit" disabled={saving} className="bg-forest hover:bg-forest-dark">
                {saving ? "Saving…" : cooperative ? "Save changes" : "Register group"}
              </Button>
            </div>
          </div>
        </form>
      </DialogContent>
    </Dialog>
  );
}
