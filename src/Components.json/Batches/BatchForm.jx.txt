import React, { useEffect, useState } from "react";
import { base44 } from "@/api/base44Client";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
import { COMMODITIES, NIGERIAN_STATES } from "@/lib/constants";
import { todayIso } from "@/lib/format";

const EMPTY = {
  commodity: "Maize",
  source_type: "farmer",
  source_name: "",
  farmer_code: "",
  state: "",
  lga: "",
  harvest_date: todayIso(),
  quantity: "",
  unit: "kg",
  aggregation_location: "",
  storage_location: "",
  processing_notes: "",
  destination: "",
  status: "registered",
  quality_status: "not_tested",
};

export default function BatchForm({ open, onOpenChange, onSaved }) {
  const [form, setForm] = useState(EMPTY);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    if (open) {
      setForm(EMPTY);
      setError("");
    }
  }, [open]);

  const set = (field) => (event) => setForm({ ...form, [field]: event.target.value });

  const submit = async (event) => {
    event.preventDefault();
    setSaving(true);
    setError("");
    try {
      const total = await base44.entities.CommodityBatch.count();
      const sequence = String(total + 1).padStart(4, "0");
      const batchCode = `NT-${form.commodity.slice(0, 2).toUpperCase()}-${sequence}`;
      const traceCode = `NT${Date.now().toString(36).toUpperCase()}`;

      await base44.entities.CommodityBatch.create({
        ...form,
        quantity: form.quantity === "" ? null : Number(form.quantity),
        batch_code: batchCode,
        trace_code: traceCode,
        demo_record: false,
      });

      await base44.entities.TraceEvent.create({
        batch_code: batchCode,
        event_type: "registered",
        event_date: form.harvest_date || todayIso(),
        location: form.aggregation_location || form.destination || "—",
        actor: form.source_name || "Field officer",
        notes: `Batch registered from ${form.source_name || "source"}.`,
        demo_record: false,
      });

      onSaved();
      onOpenChange(false);
    } catch (err) {
      setError("The batch could not be registered. Please check the required fields.");
    } finally {
      setSaving(false);
    }
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-h-[92vh] max-w-2xl overflow-y-auto">
        <DialogHeader>
          <DialogTitle className="font-display text-2xl text-forest-dark">Register a commodity batch</DialogTitle>
          <DialogDescription className="text-[12px]">
            A unique batch code and traceability code are generated on submission, and the registration event is
            written to the batch history.
          </DialogDescription>
        </DialogHeader>

        <form onSubmit={submit} className="mt-4 space-y-4">
          <div className="grid gap-3.5 sm:grid-cols-2">
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Commodity
              </Label>
              <select value={form.commodity} onChange={set("commodity")} className="select-input mt-1.5">
                {COMMODITIES.map((item) => (
                  <option key={item}>{item}</option>
                ))}
              </select>
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Source type
              </Label>
              <select value={form.source_type} onChange={set("source_type")} className="select-input mt-1.5">
                <option value="farmer">Farmer</option>
                <option value="cooperative">Cooperative</option>
                <option value="aggregator">Aggregator</option>
              </select>
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Source name
              </Label>
              <Input required className="mt-1.5" value={form.source_name} onChange={set("source_name")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Farmer ID (optional)
              </Label>
              <Input className="mt-1.5" value={form.farmer_code} onChange={set("farmer_code")} placeholder="NT-FR-0001" />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                State
              </Label>
              <select required value={form.state} onChange={set("state")} className="select-input mt-1.5">
                <option value="">Select state</option>
                {NIGERIAN_STATES.map((state) => (
                  <option key={state}>{state}</option>
                ))}
              </select>
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                LGA
              </Label>
              <Input className="mt-1.5" value={form.lga} onChange={set("lga")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Harvest date
              </Label>
              <Input type="date" className="mt-1.5" value={form.harvest_date} onChange={set("harvest_date")} />
            </div>
            <div className="grid grid-cols-2 gap-2">
              <div>
                <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                  Quantity
                </Label>
                <Input type="number" className="mt-1.5" value={form.quantity} onChange={set("quantity")} />
              </div>
              <div>
                <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                  Unit
                </Label>
                <select value={form.unit} onChange={set("unit")} className="select-input mt-1.5">
                  <option value="kg">kg</option>
                  <option value="tonne">tonne</option>
                  <option value="bag">bag</option>
                </select>
              </div>
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Aggregation location
              </Label>
              <Input className="mt-1.5" value={form.aggregation_location} onChange={set("aggregation_location")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Storage location
              </Label>
              <Input className="mt-1.5" value={form.storage_location} onChange={set("storage_location")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Batch status
              </Label>
              <select value={form.status} onChange={set("status")} className="select-input mt-1.5">
                <option value="registered">Registered</option>
                <option value="in_storage">In storage</option>
                <option value="processed">Processed</option>
                <option value="distributed">Distributed</option>
              </select>
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Destination
              </Label>
              <Input className="mt-1.5" value={form.destination} onChange={set("destination")} />
            </div>
          </div>

          <div>
            <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
              Processing notes
            </Label>
            <Textarea rows={2} className="mt-1.5 resize-none" value={form.processing_notes} onChange={set("processing_notes")} />
          </div>

          {error && <p className="text-[12px] text-destructive">{error}</p>}

          <div className="flex justify-end gap-2 pt-1">
            <Button type="button" variant="ghost" onClick={() => onOpenChange(false)}>
              Cancel
            </Button>
            <Button type="submit" disabled={saving} className="bg-forest hover:bg-forest-dark">
              {saving ? "Registering…" : "Register batch"}
            </Button>
          </div>
        </form>
      </DialogContent>
    </Dialog>
  );
}
