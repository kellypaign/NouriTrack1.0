import React, { useEffect, useState } from "react";
import { base44 } from "@/api/base44Client";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
import { COMMODITIES, NIGERIAN_STATES } from "@/lib/constants";

const EMPTY = {
  name: "",
  type: "laboratory",
  state: "",
  lga: "",
  administrator_name: "",
  administrator_email: "",
  administrator_phone: "",
  role_in_platform: "",
  commodities: [],
  accreditation: "",
  status: "active",
  partnership_date: "",
  notes: "",
};

const TYPES = [
  { value: "laboratory", label: "Laboratory" },
  { value: "cooperative", label: "Cooperative" },
  { value: "aggregator", label: "Aggregator" },
  { value: "ngo", label: "NGO" },
  { value: "government", label: "Government" },
  { value: "private_company", label: "Private company" },
  { value: "research", label: "Research" },
];

export default function OrganisationForm({ open, onOpenChange, organisation, onSaved }) {
  const [form, setForm] = useState(EMPTY);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    if (!open) return;
    setForm(organisation ? { ...EMPTY, ...organisation } : EMPTY);
    setError("");
  }, [open, organisation]);

  const set = (field) => (event) => setForm({ ...form, [field]: event.target.value });

  const toggleCommodity = (item) =>
    setForm({
      ...form,
      commodities: form.commodities?.includes(item)
        ? form.commodities.filter((value) => value !== item)
        : [...(form.commodities || []), item],
    });

  const submit = async (event) => {
    event.preventDefault();
    setSaving(true);
    setError("");
    try {
      const payload = { ...form };
      delete payload.id;
      delete payload.created_date;
      delete payload.updated_date;
      delete payload.created_by_id;
      if (organisation) await base44.entities.Organisation.update(organisation.id, payload);
      else await base44.entities.Organisation.create({ ...payload, demo_record: false });
      onSaved();
      onOpenChange(false);
    } catch (err) {
      setError("The record could not be saved. Please check the required fields.");
    } finally {
      setSaving(false);
    }
  };

  const remove = async () => {
    setSaving(true);
    await base44.entities.Organisation.delete(organisation.id);
    setSaving(false);
    onSaved();
    onOpenChange(false);
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-h-[92vh] max-w-2xl overflow-y-auto">
        <DialogHeader>
          <DialogTitle className="font-display text-2xl text-forest-dark">
            {organisation ? "Update organisation record" : "Register partner organisation"}
          </DialogTitle>
          <DialogDescription className="text-[12px]">
            Partner organisations — laboratories, cooperatives, aggregators, NGOs and government offices — form the
            institutional ecosystem that feeds NouriTrack.
          </DialogDescription>
        </DialogHeader>

        <form onSubmit={submit} className="mt-4 space-y-4">
          <div>
            <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
              Organisation type
            </Label>
            <div className="mt-2 flex flex-wrap gap-2">
              {TYPES.map((option) => (
                <button
                  key={option.value}
                  type="button"
                  onClick={() => setForm({ ...form, type: option.value })}
                  className={`rounded-lg border px-3.5 py-2 text-[12px] font-medium transition-colors ${
                    form.type === option.value
                      ? "border-forest bg-forest-tint text-forest-dark"
                      : "border-border bg-white text-foreground/70"
                  }`}
                >
                  {option.label}
                </button>
              ))}
            </div>
          </div>

          <div className="grid gap-3.5 sm:grid-cols-2">
            <div className="sm:col-span-2">
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Organisation name
              </Label>
              <Input required className="mt-1.5" value={form.name} onChange={set("name")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                State
              </Label>
              <select value={form.state} onChange={set("state")} className="select-input mt-1.5">
                <option value="">Select state</option>
                {NIGERIAN_STATES.map((state) => (
                  <option key={state}>{state}</option>
                ))}
              </select>
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                LGA
              </Label>
              <Input className="mt-1.5" value={form.lga} onChange={set("lga")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Administrator name
              </Label>
              <Input className="mt-1.5" value={form.administrator_name} onChange={set("administrator_name")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Administrator email
              </Label>
              <Input type="email" className="mt-1.5" value={form.administrator_email} onChange={set("administrator_email")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Administrator phone
              </Label>
              <Input className="mt-1.5" value={form.administrator_phone} onChange={set("administrator_phone")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Role in platform
              </Label>
              <Input
                className="mt-1.5"
                placeholder="e.g. Mycotoxin testing, aggregation, distribution"
                value={form.role_in_platform}
                onChange={set("role_in_platform")}
              />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Accreditation / reference
              </Label>
              <Input className="mt-1.5" placeholder="e.g. ISO/IEC 17025" value={form.accreditation} onChange={set("accreditation")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Partnership date
              </Label>
              <Input type="date" className="mt-1.5" value={form.partnership_date} onChange={set("partnership_date")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Status
              </Label>
              <select value={form.status} onChange={set("status")} className="select-input mt-1.5">
                <option value="active">Active</option>
                <option value="pending">Pending setup</option>
                <option value="inactive">Inactive</option>
              </select>
            </div>
          </div>

          <div>
            <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
              Commodities handled
            </Label>
            <div className="mt-2 flex flex-wrap gap-1.5">
              {COMMODITIES.map((item) => {
                const active = form.commodities?.includes(item);
                return (
                  <button
                    key={item}
                    type="button"
                    onClick={() => toggleCommodity(item)}
                    className={`rounded-full border px-3 py-1 text-[11.5px] transition-colors ${
                      active
                        ? "border-forest bg-forest text-primary-foreground"
                        : "border-border bg-white text-foreground/70 hover:border-forest/40"
                    }`}
                  >
                    {item}
                  </button>
                );
              })}
            </div>
          </div>

          <div>
            <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
              Notes
            </Label>
            <Textarea rows={2} className="mt-1.5 resize-none" value={form.notes} onChange={set("notes")} />
          </div>

          {error && <p className="text-[12px] text-destructive">{error}</p>}

          <div className="flex items-center justify-between gap-2 pt-1">
            {organisation ? (
              <Button type="button" variant="ghost" className="text-destructive" onClick={remove} disabled={saving}>
                Delete record
              </Button>
            ) : (
              <span />
            )}
            <div className="flex gap-2">
              <Button type="button" variant="ghost" onClick={() => onOpenChange(false)}>
                Cancel
              </Button>
              <Button type="submit" disabled={saving} className="bg-forest hover:bg-forest-dark">
                {saving ? "Saving…" : organisation ? "Save changes" : "Register organisation"}
              </Button>
            </div>
          </div>
        </form>
      </DialogContent>
    </Dialog>
  );
}
