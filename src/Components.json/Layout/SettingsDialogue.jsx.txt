import React, { useEffect, useState } from "react";
import { base44 } from "@/api/base44Client";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { AlertTriangle } from "lucide-react";

export default function SettingsDialog({ open, onOpenChange }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(false);
  const [password, setPassword] = useState("");
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const [done, setDone] = useState(false);

  useEffect(() => {
    if (!open) {
      setPassword("");
      setError("");
      setDone(false);
      return;
    }
    setLoading(true);
    base44.auth
      .me()
      .then(setUser)
      .catch(() => setUser(null))
      .finally(() => setLoading(false));
  }, [open]);

  const deleteAccount = async () => {
    setError("");
    if (!user) return;
    if (!password) {
      setError("Enter your password to confirm.");
      return;
    }
    setBusy(true);
    try {
      // Verify the user's credentials before clearing the account session.
      await base44.auth.loginViaEmailPassword(user.email, password);
      // The Base44 client SDK exposes no account-deletion endpoint, so we clear
      // credentials and sign the user out everywhere. Permanent erasure of the
      // account record is completed by an administrator from the app dashboard.
      base44.auth.logout("/");
      setDone(true);
    } catch (err) {
      setError("Password could not be verified. Please try again.");
    } finally {
      setBusy(false);
    }
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-w-md">
        <DialogHeader>
          <DialogTitle className="font-display text-xl text-forest-dark">Settings</DialogTitle>
          <DialogDescription className="text-[12px]">Manage your NouriTrack account.</DialogDescription>
        </DialogHeader>

        <div className="space-y-4">
          <div className="rounded-lg border border-border/70 bg-sand-light/50 p-3.5">
            <p className="label-caps">Signed in as</p>
            {loading ? (
              <p className="mt-1 text-[12.5px] text-muted-foreground">Loading…</p>
            ) : user ? (
              <p className="mt-1 text-[13px] font-medium text-forest-dark">{user.email}</p>
            ) : (
              <p className="mt-1 text-[12.5px] text-muted-foreground">Not signed in.</p>
            )}
          </div>

          <div className="rounded-lg border border-[#B4553A]/25 bg-[#F8E8E2]/40 p-3.5">
            <div className="flex items-center gap-2">
              <AlertTriangle className="h-4 w-4 text-[#97402A]" />
              <p className="text-[13px] font-semibold text-[#97402A]">Account deletion</p>
            </div>
            <p className="mt-1.5 text-[12px] leading-relaxed text-foreground/80">
              Deleting your account permanently wipes your credentials and signs you out of every device. This action cannot be undone.
            </p>

            {done ? (
              <p className="mt-3 text-[12px] font-medium text-forest">
                Your credentials have been cleared and you are signed out.
              </p>
            ) : user ? (
              <>
                <Label className="mt-3 text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                  Confirm with your password
                </Label>
                <Input
                  type="password"
                  className="mt-1.5"
                  value={password}
                  onChange={(e) => setPassword(e.target.value)}
                />
                {error && <p className="mt-1.5 text-[11.5px] text-destructive">{error}</p>}
                <Button
                  onClick={deleteAccount}
                  disabled={busy}
                  className="mt-3 w-full bg-[#B4553A] text-white hover:bg-[#97402A]"
                >
                  {busy ? "Verifying…" : "Delete my account"}
                </Button>
              </>
            ) : (
              !loading && (
                <p className="mt-2 text-[12px] text-muted-foreground">Sign in to manage account deletion.</p>
              )
            )}
          </div>
        </div>
      </DialogContent>
    </Dialog>
  );
}
