import React from "react";
import { NavLink } from "react-router-dom";
import {
  LayoutDashboard,
  Map,
  BellRing,
  Users,
  Network,
  Package,
  FlaskConical,
  HeartPulse,
  UsersRound,
  LineChart,
  Building2,
  Home,
  Leaf,
} from "lucide-react";
import { cn } from "@/lib/utils";
import { NAV_SECTIONS } from "@/lib/portals";
import { usePortal } from "@/components/layout/PortalContext";

const ICONS = {
  dashboard: LayoutDashboard,
  map: Map,
  alerts: BellRing,
  farmers: Users,
  cooperatives: Network,
  batches: Package,
  quality: FlaskConical,
  nutrition: HeartPulse,
  beneficiaries: UsersRound,
  impact: LineChart,
  stakeholders: Building2,
  household: Home,
};

export default function Sidebar({ onNavigate }) {
  const { portal } = usePortal();

  return (
    <div className="flex h-full flex-col bg-white">
      <div className="flex items-center gap-3 border-b border-border/70 px-5 py-5">
        <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-forest">
          <Leaf className="h-5 w-5 text-gold" strokeWidth={1.75} />
        </div>
        <div className="leading-tight">
          <p className="font-display text-[19px] tracking-tight text-forest-dark">NouriTrack</p>
          <p className="text-[9px] font-semibold uppercase tracking-[0.16em] text-muted-foreground">
            SciAg Agroscientific
          </p>
        </div>
      </div>

      <nav className="flex-1 overflow-y-auto px-3 py-5">
        {NAV_SECTIONS.map((section) => {
          const items = section.items.filter((item) => portal.nav.includes(item.key));
          if (!items.length) return null;
          return (
            <div key={section.label} className="mb-6">
              <p className="px-3 pb-2 text-[9px] font-semibold uppercase tracking-[0.18em] text-muted-foreground/70">
                {section.label}
              </p>
              <div className="space-y-0.5">
                {items.map((item) => {
                  const Icon = ICONS[item.icon] || LayoutDashboard;
                  return (
                    <NavLink
                      key={item.key}
                      to={item.path}
                      onClick={onNavigate}
                      className={({ isActive }) =>
                        cn(
                          "group relative flex items-center gap-3 rounded-lg px-3 py-2 text-[13px] font-medium transition-all duration-200",
                          isActive
                            ? "bg-forest-tint text-forest-dark"
                            : "text-foreground/70 hover:bg-sand-light hover:text-forest-dark"
                        )
                      }
                    >
                      {({ isActive }) => (
                        <>
                          <span
                            className={cn(
                              "absolute left-0 top-1/2 h-4 w-[2px] -translate-y-1/2 rounded-full bg-gold transition-opacity",
                              isActive ? "opacity-100" : "opacity-0"
                            )}
                          />
                          <Icon
                            className={cn(
                              "h-[16px] w-[16px] shrink-0",
                              isActive ? "text-forest" : "text-muted-foreground group-hover:text-forest"
                            )}
                            strokeWidth={1.8}
                          />
                          <span className="truncate">{item.label}</span>
                        </>
                      )}
                    </NavLink>
                  );
                })}
              </div>
            </div>
          );
        })}
      </nav>

      <div className="border-t border-border/70 px-5 py-4">
        <p className="label-caps mb-1">Signed in as</p>
        <p className="text-[12.5px] font-medium leading-snug text-forest-dark">{portal.name}</p>
        <p className="mt-0.5 text-[11px] text-muted-foreground">{portal.organisation}</p>
        <p className="mt-3 rounded-md bg-gold-light/70 px-2 py-1.5 text-[9.5px] font-semibold uppercase tracking-[0.1em] text-gold-dark">
          Demonstration data
        </p>
      </div>
    </div>
  );
}
