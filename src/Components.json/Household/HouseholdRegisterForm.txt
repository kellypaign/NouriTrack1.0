import React, { useEffect, useState } from "react";
import { base44 } from "@/api/base44Client";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
import { NIGERIAN_STATES } from "@/lib/constants";
import { todayIso } from "@/lib/format";

const suggestCode = (state) =>
  `HH-${(state || "NG").slice(0, 3).toUpperCase()}-${Math.floor(1000 + Math.random() * 9000)}`;

const EMPTY = {
  household_code: "",
  community: "",
  ward: "",
  lga: "",
  state: "",
  household_size: "",
  children_under5: "",
  children_6_23_months: "",
  school_age_children: "",
  caregiver_name: "",
  caregiver_phone: "",
  contact_consent: false,
  latitude: "",
  longitude: "",
  gps_consent: false,
  child_data_consent: false,
  registered_by: "self",
  registering_officer: "",
  notes: "",
};

export default function HouseholdRegisterForm({ open, onOpenChange, onSaved }) {
  const [form, setForm] = useState(EMPTY);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    if (!open) return;
    setForm({ ...EMPTY, household_code: suggestCode("") });
    setError("");
  }, [open]);

  const set = (field) => (event) => setForm({ ...form, [field]: event.target.value });
  const numeric = (field) => (event) =>
    setForm({ ...form, [field]: event.target.value === "" ? "" : Number(event.target.value) });

  const submit = async (event) => {
    event.preventDefault();
    if (!form.household_code || !form.community || !form.state) {
      setError("Household code, community and state are required.");
      return;
    }
    if (form.caregiver_phone && !form.contact_consent) {
      setError("Please confirm consent before sharing caregiver contact details.");
      return;
    }
    setSaving(true);
    setError("");
    try {
      const payload = {
        ...form,
        household_size: form.household_size === "" ? null : Number(form.household_size),
        children_under5: form.children_under5 === "" ? null : Number(form.children_under5),
        children_6_23_months: form.children_6_23_months === "" ? null : Number(form.children_6_23_months),
        school_age_children: form.school_age_children === "" ? null : Number(form.school_age_children),
        latitude: form.latitude === "" ? null : Number(form.latitude),
        longitude: form.longitude === "" ? null : Number(form.longitude),
      };
      delete payload.id;
      const created = await base44.entities.Household.create({ ...payload, demo_record: false });
      onSaved?.(created);
      onOpenChange(false);
    } catch (err) {
      setError("The household could not be registered. Please check the fields and try again.");
    } finally {
      setSaving(false);
    }
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-h-[92vh] max-w-2xl overflow-y-auto">
        <DialogHeader>
          <DialogTitle className="font-display text-2xl text-forest-dark">Register your household</DialogTitle>
          <DialogDescription className="text-[12px]">
            Tell us about your household and community. A field officer can also register a household on behalf of a family.
          </DialogDescription>
        </DialogHeader>

        <form onSubmit={submit} className="mt-4 space-y-4">
          <div>
            <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
              Registered by
            </Label>
            <div className="mt-2 flex gap-2">
              {[
                { value: "self", label: "Household member (self)" },
                { value: "field_officer", label: "Field officer (assisted)" },
              ].map((option) => (
                <button
                  key={option.value}
                  type="button"
                  onClick={() => setForm({ ...form, registered_by: option.value })}
                  className={`rounded-lg border px-3.5 py-2 text-[12px] font-medium transition-colors ${
                    form.registered_by === option.value
                      ? "border-forest bg-forest-tint text-forest-dark"
                      : "border-border bg-white text-foreground/70"
                  }`}
                >
                  {option.label}
                </button>
              ))}
            </div>
          </div>

          {form.registered_by === "field_officer" && (
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Registering officer name
              </Label>
              <Input className="mt-1.5" value={form.registering_officer} onChange={set("registering_officer")} />
            </div>
          )}

          <div className="grid gap-3.5 sm:grid-cols-2">
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Household code
              </Label>
              <Input required className="mt-1.5" value={form.household_code} onChange={set("household_code")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                State
              </Label>
              <select required value={form.state} onChange={set("state")} className="select-input mt-1.5">
                <option value="">Select state</option>
                {NIGERIAN_STATES.map((state) => (
                  <option key={state}>{state}</option>
                ))}
              </select>
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                LGA
              </Label>
              <Input className="mt-1.5" value={form.lga} onChange={set("lga")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Ward
              </Label>
              <Input className="mt-1.5" value={form.ward} onChange={set("ward")} />
            </div>
            <div className="sm:col-span-2">
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Community
              </Label>
              <Input required className="mt-1.5" value={form.community} onChange={set("community")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Household size
              </Label>
              <Input type="number" className="mt-1.5" value={form.household_size} onChange={numeric("household_size")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Children under 5
              </Label>
              <Input type="number" className="mt-1.5" value={form.children_under5} onChange={numeric("children_under5")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Children 6–23 months
              </Label>
              <Input
                type="number"
                className="mt-1.5"
                value={form.children_6_23_months}
                onChange={numeric("children_6_23_months")}
              />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                School-age children
              </Label>
              <Input
                type="number"
                className="mt-1.5"
                value={form.school_age_children}
                onChange={numeric("school_age_children")}
              />
            </div>
          </div>

          <div className="grid gap-3.5 sm:grid-cols-2">
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Caregiver name
              </Label>
              <Input className="mt-1.5" value={form.caregiver_name} onChange={set("caregiver_name")} />
            </div>
            <div>
              <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
                Caregiver phone
              </Label>
              <Input className="mt-1.5" value={form.caregiver_phone} onChange={set("caregiver_phone")} />
            </div>
          </div>

          <div className="space-y-2 rounded-lg border border-border bg-sand-light/60 p-3.5">
            <label className="flex items-start gap-2.5 text-[12px] leading-relaxed text-foreground/80">
              <input
                type="checkbox"
                className="mt-0.5 h-4 w-4 rounded border-border text-forest"
                checked={form.contact_consent}
                onChange={(e) => setForm({ ...form, contact_consent: e.target.checked })}
              />
              I consent to NouriTrack storing the caregiver contact details above so nutrition officers can follow up.
            </label>
            <label className="flex items-start gap-2.5 text-[12px] leading-relaxed text-foreground/80">
              <input
                type="checkbox"
                className="mt-0.5 h-4 w-4 rounded border-border text-forest"
                checked={form.child_data_consent}
                onChange={(e) => setForm({ ...form, child_data_consent: e.target.checked })}
              />
              I consent to sharing child feeding information for children in this household.
            </label>
            <label className="flex items-start gap-2.5 text-[12px] leading-relaxed text-foreground/80">
              <input
                type="checkbox"
                className="mt-0.5 h-4 w-4 rounded border-border text-forest"
                checked={form.gps_consent}
                onChange={(e) => setForm({ ...form, gps_consent: e.target.checked })}
              />
              I consent to sharing our household location (GPS) to help plan community support.
            </label>
            {form.gps_consent && (
              <div className="grid gap-3 sm:grid-cols-2">
                <Input placeholder="Latitude" value={form.latitude} onChange={numeric("latitude")} />
                <Input placeholder="Longitude" value={form.longitude} onChange={numeric("longitude")} />
              </div>
            )}
          </div>

          <div>
            <Label className="text-[11px] font-semibold uppercase tracking-[0.1em] text-muted-foreground">
              Notes
            </Label>
            <Textarea rows={2} className="mt-1.5 resize-none" value={form.notes} onChange={set("notes")} />
          </div>

          {error && <p className="text-[12px] text-destructive">{error}</p>}

          <div className="flex justify-end gap-2 pt-1">
            <Button type="button" variant="ghost" onClick={() => onOpenChange(false)}>
              Cancel
            </Button>
            <Button type="submit" disabled={saving} className="bg-forest hover:bg-forest-dark">
              {saving ? "Saving…" : "Register household"}
            </Button>
          </div>
        </form>
      </DialogContent>
    </Dialog>
  );
}
