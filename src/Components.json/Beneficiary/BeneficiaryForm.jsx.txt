import React, { useEffect, useState } from "react";
import { base44 } from "@/api/base44Client";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
import { todayIso } from "@/lib/format";

const EMPTY = {
  household_code: "",
  community: "",
  state: "",
  lga: "",
  category: "household",
  programme: "",
  enrolled_date: todayIso(),
  status: "enrolled",
  members_count: "",
  children_under5: "",
  follow_up_date: "",
  notes: "",
};

export default function BeneficiaryForm({ open, onOpenChange, beneficiary, onSaved }) {
  const [form, setForm] = useState(EMPTY);
  const [communities, setCommunities] = useState([]);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    if (!open) return;
    setForm(beneficiary ? { ...EMPTY, ...beneficiary } : { ...EMPTY, household_code: `HH-${Date.now().toString().slice(-6)}` });
    setError("");
    base44.entities.Community.filter({}, { limit: 100 }).then((page) => setCommunities(page?.items || []));
  }, [open, beneficiary]);

  const set = (field) => (event) => setForm({ ...form, [field]: event.target.value });
  const num = (field) => (event) =>
    setForm({ ...form, [field]: event.target.value === "" ? "" : Number(event.target.value) });

  const chooseCommunity = (name) => {
    const match = communities.find((community) => community.name === name);
    setForm({
      ...form,
      community: name,
      state: match?.state || form.state,
      lga: match?.lga || form.lga,
      programme: form.programme || match?.programme || "",
    });
  };

  const submit = async (event) => {
    event.preventDefault();
    setSaving(true);
    setError("");
    try {
      if (!beneficiary && form.household_code) {
        const existing = await base44.entities.Beneficiary.filter({ household_code: form.household_code });
        if ((existing?.items || []).length > 0) {
          setError("This household code is already enrolled. Duplicate registration is not permitted.");
          setSaving(false);
          return;
        }
      }

      const payload = {
        ...form,
        members_count: form.members_count === "" ? null : Number(form.members_count),
        children_under5: form.children_under5 === "" ? null : Number(form.children_under5),
      };
      delete payload.id;
      delete payload.created_date;
      delete payload.updated_date;
      delete payload.created_by_id;

      if (beneficiary) await base44.entities.Beneficiary.update(beneficiary.id, payload);
      else await base44.entities.Beneficiary.create({ ...payload, demo_record: false });
      onSaved();
      onOpenChange(false);
    } catch (err) {
      setError("The beneficiary record could not be saved.");
    } finally {
      setSaving(false);
    }
  };

  const remove = async () => {
    setSaving(true);
    await base44.entities.Beneficiary.delete(beneficiary.id);
    setSaving(false);
    onSaved();
    onOpenChange(false);
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-h-[92vh] max-w-xl overflow-y-auto">
        <DialogHeader>
          <DialogTitle className="font-display text-2xl text-forest-dark">
            {beneficiary ? "Update household record" : "Enrol household"}
          </DialogTitle>
          <DialogDescription className="text-[12px]">
            Households are identified by a code, never by name. Use the household code issued by your programme
            to prevent duplicate registration.
          </DialogDescription>
        </DialogHeader>

        <form onSubmit={submit} className="mt-4 space-y-4">
          <div className="grid gap-3.5 sm:grid-cols-2">
            <div>
              <Label className="field-label">Household code</Label>
              <Input required className="mt-1.5 font-mono text-[12px]" value={form.household_code} onChange={set("household_code")} />
            </div>
            <div>
              <Label className="field-label">Community</Label>
              <select required value={form.community} onChange={(e) => chooseCommunity(e.target.value)} className="select-input mt-1.5">
                <option value="">Select community</option>
                {communities.map((community) => (
                  <option key={community.id} value={community.name}>
                    {community.name}
                  </option>
                ))}
              </select>
            </div>
            <div>
              <Label className="field-label">LGA</Label>
              <Input className="mt-1.5" value={form.lga} onChange={set("lga")} />
            </div>
            <div>
              <Label className="field-label">State</Label>
              <Input className="mt-1.5" value={form.state} onChange={set("state")} />
            </div>
            <div>
              <Label className="field-label">Beneficiary category</Label>
              <select value={form.category} onChange={set("category")} className="select-input mt-1.5">
                <option value="household">Household</option>
                <option value="child_under5">Child under 5</option>
                <option value="school_age_child">School-age child</option>
                <option value="pregnant_or_lactating">Pregnant or lactating woman</option>
                <option value="other">Other</option>
              </select>
            </div>
            <div>
              <Label className="field-label">Programme</Label>
              <Input className="mt-1.5" value={form.programme} onChange={set("programme")} />
            </div>
            <div>
              <Label className="field-label">Enrolment date</Label>
              <Input type="date" className="mt-1.5" value={form.enrolled_date} onChange={set("enrolled_date")} />
            </div>
            <div>
              <Label className="field-label">Status</Label>
              <select value={form.status} onChange={set("status")} className="select-input mt-1.5">
                <option value="enrolled">Enrolled</option>
                <option value="active">Active</option>
                <option value="completed">Completed</option>
                <option value="withdrawn">Withdrawn</option>
              </select>
            </div>
            <div>
              <Label className="field-label">Household members</Label>
              <Input type="number" className="mt-1.5" value={form.members_count} onChange={num("members_count")} />
            </div>
            <div>
              <Label className="field-label">Children under 5</Label>
              <Input type="number" className="mt-1.5" value={form.children_under5} onChange={num("children_under5")} />
            </div>
            <div>
              <Label className="field-label">Next follow-up</Label>
              <Input type="date" className="mt-1.5" value={form.follow_up_date} onChange={set("follow_up_date")} />
            </div>
          </div>

          <div>
            <Label className="field-label">Follow-up notes</Label>
            <Textarea rows={2} className="mt-1.5 resize-none" value={form.notes} onChange={set("notes")} />
          </div>

          {error && <p className="text-[12px] text-destructive">{error}</p>}

          <div className="flex items-center justify-between gap-2 pt-1">
            {beneficiary ? (
              <Button type="button" variant="ghost" className="text-destructive" onClick={remove} disabled={saving}>
                Delete record
              </Button>
            ) : (
              <span />
            )}
            <div className="flex gap-2">
              <Button type="button" variant="ghost" onClick={() => onOpenChange(false)}>
                Cancel
              </Button>
              <Button type="submit" disabled={saving} className="bg-forest hover:bg-forest-dark">
                {saving ? "Saving…" : beneficiary ? "Save changes" : "Enrol household"}
              </Button>
            </div>
          </div>
        </form>
      </DialogContent>
    </Dialog>
  );
}
